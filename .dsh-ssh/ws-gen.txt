/** 執行工作流。emit canvas 事件；text/config 節點視為完成，image/video 調 runNode。 */
  async generate(workflowId: string): Promise<{ ok: boolean; status: string }> {
    const wf = this.workflows.get(workflowId)
    if (!wf) return { ok: false, status: "workflow not found" }
    wf.status = "running"

    // topo 排序；有循環 → 報錯
    const { order, cycle } = topologicalSort(wf.graph)
    if (cycle.length) {
      wf.status = "failed"
      return { ok: false, status: `cycle: ${cycle.join(",")}` }
    }

    // 展開節點流 → emit plan + node.created + connected（前端渲染整條流）
    this.emitGraph(wf)

    // 逐節點執行
    const upstreamContent = new Map<string, string>()
    for (const nodeId of order) {
      const node = wf.graph.nodes.find((n) => n.id === nodeId)
      if (!node) continue
      if (node.kind === "text" || node.kind === "config") continue

      wf.nodeStatus[nodeId] = "running"
      this.opts.emit(wf.owner, "canvas.node.generating", { unit_id: workflowId, node_id: nodeId, status: "running" })
      const result = await this.opts.runNode(node, upstreamContent, wf.graph)
      if (result.ok && result.content) {
        wf.nodeStatus[nodeId] = "success"
        wf.nodeContent[nodeId] = result.content
        if (result.costCoins) wf.nodeCost[nodeId] = result.costCoins
        upstreamContent.set(nodeId, result.content)
        this.opts.emit(wf.owner, "canvas.node.updated", {
          unit_id: workflowId,
          node_id: nodeId,
          patch: { status: "success", content: result.content, ...(result.costCoins ? { cost_coins: result.costCoins } : {}) },
          url: result.content,
          cost_coins: result.costCoins,
        })
      } else {
        wf.nodeStatus[nodeId] = "error"
        wf.status = "failed"
        this.opts.emit(wf.owner, "canvas.node.updated", { unit_id: workflowId, node_id: nodeId, patch: { status: "error", errorDetails: result.error } })
      }
    }

    const failed = Object.values(wf.nodeStatus).some((s) => s === "error")
    wf.status = failed ? "failed" : "success"
    return { ok: !failed, status: wf.status }
  }

  status(workflowId: string) {
    const wf = this.workflows.get(workflowId)
    if (!wf) return null
    return {
      workflowId: wf.workflowId,
      templateId: wf.templateId,
      status: wf.status,
      nodes: wf.graph.nodes.map((n) => ({
        id: n.id,
        role: n.role,
        kind: n.kind,
        title: n.title,
        status: wf.nodeStatus[n.id],
        content: wf.nodeContent[n.id],
        costCoins: wf.nodeCost[n.id],
      })),
      connections: wf.graph.connections,
    }
  }

  /** emit 整條節點流（plan + 節點 + 連線），前端一氣渲染。 */
  private emitGraph(wf: ActiveWorkflow): void {
    this.opts.emit(wf.owner, "canvas.plan", { unit_id: wf.workflowId, tool: `workflow:${wf.templateId}`, label: `工作流 ${wf.templateId}`, prompt: "", model: "", size: "", count: 1, kind: "workflow", source_image: null, cost_hint: null })
